 236:         for stat_name, func in [("mean_across_folds", "mean"), ("sd_across_folds", "std")]:

 239:                 "evaluation": stat_name,
 240:                 "n": int(g["test_n"].sum()),
 241:             }
 242:             for c in ["RMSE_Mg_ha", "MAE_Mg_ha", "R2", "Bias_Mg_ha",

 247:                     row[c] = float(g[c].std(ddof=1))
