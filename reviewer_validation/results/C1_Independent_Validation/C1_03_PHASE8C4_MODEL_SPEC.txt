 140: def metric_row(y, p, model, fold=None, train_n=None, test_n=None):
 141:     y = np.asarray(y, float)
 142:     p = np.asarray(p, float)
 143:     rmse = float(np.sqrt(mean_squared_error(y, p)))
 144:     mae = float(mean_absolute_error(y, p))
 145:     r2 = float(r2_score(y, p))
 146:     bias = float(np.mean(p - y))
 147:     mean_y = float(np.mean(y))
 148:     rrmse = float(rmse / mean_y * 100.0) if mean_y != 0 else np.nan
 149: 
 150:     if len(y) >= 3 and np.std(y) > 0 and np.std(p) > 0:

 154:         pr = np.nan
 155:         sr = np.nan

 158:     pred_sd = float(np.std(p, ddof=1))

 161:         "model": model,

 163:         "n": len(y),
