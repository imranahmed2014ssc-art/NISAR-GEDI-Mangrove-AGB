  82: def sha256_file(path, chunk_size=8 * 1024 * 1024):
  83:     h = hashlib.sha256()
  84:     with open(path, "rb") as f:
  85:         for chunk in iter(lambda: f.read(chunk_size), b""):
  86:             h.update(chunk)
  87:     return h.hexdigest()

  89: def numeric_complete(df, cols):
  90:     x = df[cols].apply(pd.to_numeric, errors="coerce")
  91:     if x.isna().any().any():
  92:         bad = x.columns[x.isna().any()].tolist()
  93:         raise RuntimeError(f"Missing/non-numeric values in predictors: {bad}")
  94:     return x

  96: def vif_table(X):
  97:     """
  98:     VIF-like redundancy diagnostic using each standardized predictor as response
  99:     and all other standardized predictors as covariates.
 100:     This uses predictor values only; AGBD is not involved.
 101:     """
 102:     scaler = StandardScaler()
 103:     Z = scaler.fit_transform(X.to_numpy(float))
 104:     rows = []
 105:     for j, name in enumerate(X.columns):
 106:         y = Z[:, j]
 107:         idx = [k for k in range(Z.shape[1]) if k != j]
 108:         reg = LinearRegression().fit(Z[:, idx], y)
 109:         r2 = float(reg.score(Z[:, idx], y))
 110:         if r2 >= 1.0 - 1e-12:
 111:             vif = np.inf
 112:         else:
 113:             vif = 1.0 / (1.0 - r2)
 114:         rows.append({
 115:             "predictor": name,
 116:             "R2_explained_by_other_predictors": r2,
 117:             "VIF_like": vif,
 118:         })
 119:     return pd.DataFrame(rows).sort_values(
 120:         ["VIF_like", "R2_explained_by_other_predictors"],
 121:         ascending=False
 122:     )

 124: def high_corr_pairs(corr, method):
 125:     rows = []
 126:     cols = list(corr.columns)
 127:     for i in range(len(cols)):
 128:         for j in range(i + 1, len(cols)):
 129:             r = float(corr.iloc[i, j])
 130:             ar = abs(r)
 131:             if ar >= HIGH_RHO:
 132:                 rows.append({
 133:                     "method": method,
 134:                     "predictor_1": cols[i],
 135:                     "predictor_2": cols[j],
 136:                     "correlation": r,
 137:                     "abs_correlation": ar,
 138:                     "redundancy_level": (
 139:                         "VERY_HIGH_>=0.95" if ar >= VERY_HIGH_RHO
 140:                         else "HIGH_0.85_to_<0.95"
 141:                     ),
 142:                 })
 143:     return pd.DataFrame(rows).sort_values(
 144:         "abs_correlation", ascending=False
 145:     ) if rows else pd.DataFrame(
 146:         columns=[
 147:             "method", "predictor_1", "predictor_2",
 148:             "correlation", "abs_correlation", "redundancy_level"
 149:         ]
 150:     )

 152: def agbd_descriptive(df, predictors):
 153:     """
 154:     Descriptive only. These results MUST NOT be used to choose predictors from
 155:     the full dataset before outer spatial CV.
 156:     """
 157:     y = pd.to_numeric(df["agbd"], errors="coerce").to_numpy(float)
 158:     rows = []
 159:     for p in predictors:
 160:         if p not in df.columns:
 161:             continue
 162:         x = pd.to_numeric(df[p], errors="coerce").to_numpy(float)
 163:         m = np.isfinite(x) & np.isfinite(y)
 164:         if m.sum() < 3:
 165:             continue
 166:         pe = pearsonr(x[m], y[m])
 167:         sp = spearmanr(x[m], y[m])
 168:         rows.append({
 169:             "predictor": p,
 170:             "n": int(m.sum()),
 171:             "Pearson_r_with_AGBD_DESCRIPTIVE_ONLY": float(pe.statistic),
 172:             "Spearman_rho_with_AGBD_DESCRIPTIVE_ONLY": float(sp.statistic),
 173:         })
 174:     return pd.DataFrame(rows)

 176: def variance_table(X):
 177:     rows = []
 178:     for p in X.columns:
 179:         v = X[p].to_numpy(float)
 180:         sd = float(np.std(v, ddof=1))
 181:         rows.append({
 182:             "predictor": p,
 183:             "mean": float(np.mean(v)),
 184:             "sd": sd,
 185:             "cv_abs": float(abs(sd / np.mean(v))) if np.mean(v) != 0 else np.nan,
 186:             "min": float(np.min(v)),
 187:             "median": float(np.median(v)),
 188:             "max": float(np.max(v)),
 189:             "near_zero_variance": bool(sd < 1e-8),
 190:         })
 191:     return pd.DataFrame(rows)

 193: def plot_corr_heatmap(corr, path_png, path_pdf, title):
 194:     # Reorder with hierarchical clustering using 1-|rho| as distance.
 195:     abs_corr = np.abs(corr.to_numpy(float))
 196:     dist = 1.0 - abs_corr
 197:     np.fill_diagonal(dist, 0.0)
 198: 
 199:     if len(corr) > 2:
 200:         condensed = squareform(dist, checks=False)
 201:         Z = linkage(condensed, method="average")
 202:         order = leaves_list(Z)
 203:     else:
 204:         order = np.arange(len(corr))
 205: 
 206:     ordered = corr.iloc[order, order]
 207: 
 208:     fig, ax = plt.subplots(figsize=(11.5, 9.5))
 209:     im = ax.imshow(ordered.to_numpy(float), vmin=-1, vmax=1, aspect="auto")
 210:     ax.set_xticks(np.arange(len(ordered.columns)))
 211:     ax.set_yticks(np.arange(len(ordered.index)))
 212:     ax.set_xticklabels(ordered.columns, rotation=70, ha="right", fontsize=8)
 213:     ax.set_yticklabels(ordered.index, fontsize=8)
 214:     ax.set_title(title, pad=12)
 215:     cbar = fig.colorbar(im, ax=ax, fraction=0.035, pad=0.03)
 216:     cbar.set_label("Correlation")
 217:     fig.tight_layout()
 218:     fig.savefig(path_png, dpi=450, bbox_inches="tight")
 219:     fig.savefig(path_pdf, bbox_inches="tight")
 220:     plt.close(fig)
